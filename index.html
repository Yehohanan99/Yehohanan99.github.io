A small example of the custom code I write: send a signed webhook when an order completes.

add_action( 'woocommerce_order_status_completed', function ( $order_id ) {
    $order   = wc_get_order( $order_id );
    $payload = wp_json_encode( [ 'order' => $order_id, 'email' => $order->get_billing_email() ] );
    wp_remote_post( FULFIL_URL, [
        'headers' => [ 'X-Signature' => hash_hmac( 'sha256', $payload, FULFIL_SECRET ) ],
        'body'    => $payload,
    ] );
} );
